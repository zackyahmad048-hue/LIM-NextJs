## Draft Tighten Migration (new migration, after preconditions)

```sql
-- DRAFT - creates FORCE RLS + removes unset-scope fallback.
-- Apply only after lim_app runtime switch and integration tests pass.

DO $$
DECLARE t text;
DECLARE tables text[] := ARRAY[
  'Letter','Penugasan','Lembaga','LaporanMingguan','PermohonanGuruBantu',
  'EvaluasiKasus','EvaluasiInduk','TitikSafari','DelegasiSafari','SekolahMitra',
  'PenugasanPesantren','DaerahBinaan','KontenDakwah'
];
BEGIN
  FOREACH t IN ARRAY tables LOOP
    EXECUTE format('ALTER TABLE %I FORCE ROW LEVEL SECURITY', t);
    EXECUTE format('DROP POLICY IF EXISTS %I ON %I', t || '_tenant_select', t);
    EXECUTE format('DROP POLICY IF EXISTS %I ON %I', t || '_tenant_insert', t);
    EXECUTE format('DROP POLICY IF EXISTS %I ON %I', t || '_tenant_update', t);
    EXECUTE format('DROP POLICY IF EXISTS %I ON %I', t || '_tenant_delete', t);

    EXECUTE format($p$
      CREATE POLICY %I ON %I FOR SELECT USING (
        current_setting('app.scope', true) = 'PUSAT'
        OR ("unitId" IS NOT NULL AND "unitId" = current_setting('app.unit_id', true))
        OR (current_setting('app.scope', true) = 'SELF' AND "createdById" = current_setting('app.user_id', true))
      )$p$, t || '_tenant_select', t);

    EXECUTE format($p$
      CREATE POLICY %I ON %I FOR INSERT WITH CHECK (
        current_setting('app.scope', true) = 'PUSAT'
        OR ("unitId" IS NOT NULL AND "unitId" = current_setting('app.unit_id', true))
      )$p$, t || '_tenant_insert', t);

    EXECUTE format($p$
      CREATE POLICY %I ON %I FOR UPDATE USING (
        current_setting('app.scope', true) = 'PUSAT'
        OR ("unitId" IS NOT NULL AND "unitId" = current_setting('app.unit_id', true))
        OR (current_setting('app.scope', true) = 'SELF' AND "createdById" = current_setting('app.user_id', true))
      ) WITH CHECK (
        current_setting('app.scope', true) = 'PUSAT'
        OR ("unitId" IS NOT NULL AND "unitId" = current_setting('app.unit_id', true))
      )$p$, t || '_tenant_update', t);

    EXECUTE format($p$
      CREATE POLICY %I ON %I FOR DELETE USING (
        current_setting('app.scope', true) = 'PUSAT'
        OR ("unitId" IS NOT NULL AND "unitId" = current_setting('app.unit_id', true))
      )$p$, t || '_tenant_delete', t);
  END LOOP;
END $$;
```

> Note: the SELF clause assumes `createdById` exists. Tables without `createdById` must drop that clause; verify schema before applying.
